! ============================================================
! R4  (2911, right HSRP router - the BACKUP)
! g0/0 -> R2 g0/1 (up/WAN = NAT outside)
! g0/1 -> SW_H2   (down/LAN = HSRP + NAT inside)
! identical shape to R3 - only IPs differ + NO priority line
! (default priority 100 < R3's 110, so R4 stays Standby)
! ============================================================
enable
configure terminal
hostname R4
!
interface g0/0
 ip address 30.0.0.6 255.255.255.252
 ip nat outside
 no shutdown
exit
!
! ----- task 6: HSRP (backup side) -----
interface g0/1
 ip address 192.168.30.3 255.255.255.0
 standby 1 ip 192.168.30.1
 standby 1 preempt
 ip nat inside
 no shutdown
exit
!
! ----- task 7: static NAT (same mappings as R3, so failover keeps working)
ip nat inside source static 192.168.30.10 40.0.0.10
ip nat inside source static 192.168.30.11 40.0.0.11
!
! ----- task 8: default route up to R2 -----
ip route 0.0.0.0 0.0.0.0 30.0.0.5
!
! ----- task 13: same ACL as R3 (RENAME to YOUR last name!) -----
! must be on BOTH HSRP routers - otherwise a failover to R4 would
! bypass the restriction
ip access-list extended PILPANI
 deny ip 192.168.30.0 0.0.0.255 host 61.0.0.10
 deny ip 192.168.30.0 0.0.0.255 host 71.0.0.10
 permit ip any any
exit
interface g0/1
 ip access-group PILPANI in
exit
!
end
write memory
