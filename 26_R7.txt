! ============================================================
! R7  (2911, right VPN router, server S2 below)
! g0/0 -> R5 g0/1 (up/WAN = NAT outside + tunnel source)
! g0/1 -> S2     (down/LAN = NAT inside)
! exact mirror of R6
! ============================================================
enable
configure terminal
hostname R7
!
interface g0/0
 ip address 60.0.0.6 255.255.255.252
 ip nat outside
 no shutdown
interface g0/1
 ip address 192.168.70.1 255.255.255.0
 ip nat inside
 no shutdown
exit
!
! ----- task 11: default route up to R5 -----
ip route 0.0.0.0 0.0.0.0 60.0.0.5
!
! ----- task 9: GRE tunnel (mirror of R6) -----
interface tunnel 0
 ip address 172.16.0.2 255.255.255.252
 tunnel source g0/0
 tunnel destination 60.0.0.2
 no shutdown
exit
ip route 192.168.60.0 255.255.255.0 172.16.0.1
!
! ----- task 10: static NAT for S2 -----
ip nat inside source static 192.168.70.10 71.0.0.10
!
end
write memory
