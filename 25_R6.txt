! ============================================================
! R6  (2911, left VPN router, server S1 below)
! g0/0 -> R5 g0/0 (up/WAN = NAT outside + tunnel source)
! g0/1 -> S1     (down/LAN = NAT inside)
! covers exam tasks 9 (GRE), 10 (static NAT), 11 (default route)
! ============================================================
enable
configure terminal
hostname R6
!
interface g0/0
 ip address 60.0.0.2 255.255.255.252
 ip nat outside
 no shutdown
interface g0/1
 ip address 192.168.60.1 255.255.255.0
 ip nat inside
 no shutdown
exit
!
! ----- task 11: default route up to R5 -----
! (do this BEFORE the tunnel - the tunnel needs it to reach R7's WAN IP)
ip route 0.0.0.0 0.0.0.0 60.0.0.1
!
! ----- task 9: GRE tunnel -----
! a virtual point-to-point cable THROUGH the public network:
!   source/destination = the real WAN addresses
!   172.16.x = addresses living ON the virtual cable
! then route the REMOTE private LAN into the tunnel
interface tunnel 0
 ip address 172.16.0.1 255.255.255.252
 tunnel source g0/0
 tunnel destination 60.0.0.6
 no shutdown
exit
ip route 192.168.70.0 255.255.255.0 172.16.0.2
!
! ----- task 10: static NAT for S1 -----
ip nat inside source static 192.168.60.10 61.0.0.10
!
end
write memory
!
! verify: from S1 ping 192.168.70.10 (goes through the tunnel)
!         show interface tunnel 0 (must be up/up)
