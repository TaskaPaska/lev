! ============================================================
! R3  (2911, left HSRP router - the PRIMARY)
! g0/0 -> R2 g0/0 (up/WAN = NAT outside)
! g0/1 -> SW_H1   (down/LAN = HSRP + NAT inside)
! covers exam tasks 6 (HSRP), 7 (static NAT), 8 (default), 13 (ACL)
! ============================================================
enable
configure terminal
hostname R3
!
interface g0/0
 ip address 30.0.0.2 255.255.255.252
 ip nat outside
 no shutdown
exit
!
! ----- task 6: HSRP -----
! real IP .2, shared virtual IP .1 (hosts use .1 as gateway)
! priority 110 > R4's default 100 -> R3 is Active
! preempt = take the Active role back after recovering from a failure
interface g0/1
 ip address 192.168.30.2 255.255.255.0
 standby 1 ip 192.168.30.1
 standby 1 priority 110
 standby 1 preempt
 ip nat inside
 no shutdown
exit
!
! ----- task 7: static NAT (permanent 1:1, works in both directions) -----
ip nat inside source static 192.168.30.10 40.0.0.10
ip nat inside source static 192.168.30.11 40.0.0.11
!
! ----- task 8: default route up to R2 -----
ip route 0.0.0.0 0.0.0.0 30.0.0.1
!
! ----- task 13: named ACL (RENAME to YOUR last name!) -----
! applied INBOUND on the LAN port: we match the hosts' real 192.168.30.x
! sources BEFORE NAT rewrites them. We deny the VPN servers' GLOBAL
! addresses (that's what a host would actually try to reach).
! "permit ip any any" is mandatory - without it the invisible
! deny-everything at the end of every ACL kills ALL traffic.
ip access-list extended PILPANI
 deny ip 192.168.30.0 0.0.0.255 host 61.0.0.10
 deny ip 192.168.30.0 0.0.0.255 host 71.0.0.10
 permit ip any any
exit
interface g0/1
 ip access-group PILPANI in
exit
!
end
write memory
!
! verify HSRP: show standby brief   (R3 = Active, R4 = Standby)
! verify ACL:  ping 61.0.0.10 from ServerA -> must FAIL
!              ping 40.0.0.11 from ServerA -> must work
