! ============================================================
! SW_A1  (2960, bottom-LEFT switch: PC + laptop, VLAN 10)
! fa0/24 -> SW_D fa0/1    fa0/23 -> SW_A2 fa0/23 (redundant link)
! fa0/1 -> PC    fa0/2 -> laptop
! the redundant fa0/23 link: STP handles the loop by itself, no config
! ============================================================
enable
configure terminal
hostname SW_A1
!
vlan 10
 name USERS
vlan 20
 name SERVERS
exit
!
interface fa0/24
 switchport mode trunk
interface fa0/23
 switchport mode trunk
exit
!
interface range fa0/1 - 2
 switchport mode access
 switchport access vlan 10
exit
!
end
write memory
