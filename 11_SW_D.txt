! ============================================================
! SW_D  (2960, middle "distribution" switch under Sw_1)
! g0/1 -> Sw_1 g0/2     fa0/1 -> SW_A1 fa0/24     fa0/2 -> SW_A2 fa0/24
! job: just pass both VLANs through (all ports = trunks)
! ============================================================
enable
configure terminal
hostname SW_D
!
vlan 10
 name USERS
vlan 20
 name SERVERS
exit
!
interface g0/1
 switchport mode trunk
interface fa0/1
 switchport mode trunk
interface fa0/2
 switchport mode trunk
exit
!
end
write memory
