! ============================================================
! R2  (2911, top-center - the core)
! !! FIRST: power off -> add HWIC-2T module -> power on
! g0/0 -> R3 g0/0     g0/1 -> R4 g0/0
! g0/2 -> R1 g0/1     s0/0/0 -> R5 s0/0/0 (Serial DCE cable)
! covers exam tasks 8 (static routes to R3/R4) and 12 (BGP)
! ============================================================
enable
configure terminal
hostname R2
!
interface g0/0
 ip address 30.0.0.1 255.255.255.252
 no shutdown
interface g0/1
 ip address 30.0.0.5 255.255.255.252
 no shutdown
interface g0/2
 ip address 20.0.0.2 255.255.255.252
 no shutdown
interface s0/0/0
 ip address 50.0.0.1 255.255.255.252
 clock rate 64000
 no shutdown
exit
! (clock rate only works on the DCE side - the end with the clock icon.
!  if the icon is on R5's end, put the clock rate line there instead)
!
! ----- task 8: route to the HSRP global net -----
! primary via R3; backup via R4 with distance 50 ("floating static":
! it only activates if the R3 route dies, because 50 > default 1)
ip route 40.0.0.0 255.255.255.0 30.0.0.2
ip route 40.0.0.0 255.255.255.0 30.0.0.6 50
!
! ----- task 12: BGP to R5 -----
! you invent the AS numbers: R2 = AS 100, R5 = AS 200
! "redistribute static" = the "merge static routes" the task asks for
! the "network 20.0.0.0" line is critical: that's the address your VLAN
! hosts hide behind (NAT) - without it R5's side can't send replies back
router bgp 100
 neighbor 50.0.0.2 remote-as 200
 redistribute static
 network 20.0.0.0 mask 255.255.255.252
exit
!
end
write memory
!
! verify: show ip route   (look for B = BGP routes to 61.x and 71.x)
